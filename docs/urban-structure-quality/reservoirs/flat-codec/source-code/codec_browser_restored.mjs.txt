import { createRequire } from 'node:module';
import { readFile,writeFile,mkdir } from 'node:fs/promises';
import path from 'node:path';
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { INSPECTION_VIEWS } from '../../../../../lib/city/inspection.ts';
const require=createRequire(import.meta.url);const {chromium}=require(path.resolve('work/urban-structure/browser/node_modules/playwright'));
const root=process.cwd(),out=path.resolve('work/urban-structure/p5/full-city-review/budget-reduction/browser-restored');await mkdir(out,{recursive:true});
const baseline=path.resolve('work/urban-structure/p5/reservoirs/integration/budget/summary-fix-staging');
const candidate=path.resolve('work/urban-structure/p5/full-city-review/budget-reduction/flat-buildings-candidate');
const views=['overview','qingxiu-lake','waterfront-bridgehead'].map(id=>({id,camera:[...INSPECTION_VIEWS[id].camera.position,...INSPECTION_VIEWS[id].camera.target]}));
const originalViews=JSON.parse(await readFile('work/urban-structure/p5/full-city-review/views.json','utf8'));views.push(originalViews.views.at(-1));
const browser=await chromium.launch({headless:true,executablePath:'/Applications/Google Chrome.app/Contents/MacOS/Google Chrome'});const records=[],errors=[];
try{
 for(const version of ['before','after'])for(const profile of ['detail','smooth']){
  const context=await browser.newContext({viewport:{width:1440,height:1140},deviceScaleFactor:1});const page=await context.newPage();const assets={};
  page.on('pageerror',e=>errors.push({version,profile,message:e.message}));page.on('console',m=>{if(m.type()==='error')errors.push({version,profile,message:m.text()});});
  await page.route(/\/(models\/nanning-city(?:-mobile)?\.glb|data\/(?:overview|landmarks)\.json)(?:\?|$)/,async route=>{
   const url=new URL(route.request().url()).pathname.replace(/^.*(?=\/(?:models|data)\/)/,'');const file=path.join(version==='before'?baseline:candidate,'public'+url);const body=await readFile(file);assets[url]={file,sha256:createHash('sha256').update(body).digest('hex')};await route.fulfill({body,contentType:url.endsWith('.glb')?'model/gltf-binary':'application/json'});
  });
  for(const view of views){
   for(const material of ['clay','lit']){
    await page.goto(`http://localhost:3000/inspect?quality=${profile}&camera=${view.camera.join(',')}&hour=14&material=${material}`,{waitUntil:'networkidle'});
    const stage=page.locator('[data-testid="inspection-stage"][data-ready="true"]');await stage.waitFor({timeout:90000});await page.waitForTimeout(600);
    const id=`${version}-${profile}-${view.id}-${material}`,pending=page.waitForEvent('download');await page.getByRole('button',{name:'导出参数 JSON',exact:true}).click();await (await pending).saveAs(path.join(out,id+'.json'));const metadata=JSON.parse(await readFile(path.join(out,id+'.json'),'utf8'));
    assert.equal(metadata.settings.quality,profile);assert.equal(metadata.settings.material,material);assert.equal(metadata.canvas.frozenTime,0);assert.equal(metadata.canvas.pixelRatio,1);
    await stage.screenshot({path:path.join(out,id+'.png')});records.push({id,view,version,profile,material,assets:{...assets},metadata});await writeFile(path.join(out,'report.json'),JSON.stringify({records,errors,status:'screenshots captured; visual review separate'},null,2)+'\n');
   }
   console.log(version,profile,view.id);
  }
  await context.close();
 }
 assert.deepEqual(errors,[]);
}finally{await browser.close();}
